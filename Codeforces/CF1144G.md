### CF1144G Two Merged Sequences
---

**Rated**：2400

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10, INF = 2e9;
int f[N][5]; 
int pre[N][5];
bool ans[N];
int a[N];
int main()
{
	int n; cin >> n;
	for (int i = 1; i <= n; i ++)
		cin >> a[i];
	f[1][0] = INF, f[1][1] = -INF;
	for (int i = 2; i <= n; i ++)
	{
		f[i][0] = -INF, f[i][1] = INF;//保证一定会被更新 
		if (a[i - 1] < a[i])//a[i], a[i-1]均在 IS
		{
			if (f[i][0] < f[i - 1][0])
			{	
				f[i][0] = f[i - 1][0];
				pre[i][0] = 0;
			}
		}
		if (f[i - 1][1] < a[i])//a[i-1] DS，a[i] IS
		{
			if (f[i][0] < a[i - 1])
			{	
				f[i][0] = a[i - 1];
				pre[i][0] = 1;
			}
		}
		if (a[i - 1] > a[i])//a[i], a[i-1]均在 DS
		{
			if (f[i][1] > f[i - 1][1]) 
			{
				f[i][1] = f[i - 1][1];
				pre[i][1] = 1;	
			}
		}
		if (f[i - 1][0] > a[i])//a[i-1] IS，a[i] DS
		{
			if (f[i][1] > a[i - 1])
			{
				f[i][1] = a[i - 1];
				pre[i][1] = 0;
			}
		}
	}
	int pos = -1;
	if (f[n][0] != -INF) pos = 0;//n 被置于 IS 中 
	else if (f[n][1] != INF) pos = 1;//n 被置于 DS 中 
	if (pos == -1)
	{
		cout << "NO";
		return 0;
	}
	cout << "YES\n";
	for (int i = n; i >= 1; i --)
	{
		ans[i] = pos;
		pos = pre[i][pos];
	}
	for (int i = 1; i <= n; i ++)
		cout << ans[i] << ' ';
 	return 0;
}
```
