### P1219 [USACO1.5] 八皇后 Checker Challenge
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
int n, ans;
int a[N];
bool col[N], lg[N], ulg[N]; 
void dfs(int u)
{
    if (u > n)
    {
        ans ++;
        if (ans > 3)
            return;
        for (int i = 1; i <= n; i++)
            cout << a[i] << ' ';
        cout << '\n';
        return;
    }
    for (int i = 1; i <= n; i ++)
    {
        if (!col[i] && !lg[u + i - 1] && !ulg[n - u + i - 1])
        {
            col[i] = lg[u + i - 1] = ulg[n - u + i - 1] = 1;
            a[u] = i;
            dfs(u + 1);
            col[i] = lg[u + i - 1] = ulg[n - u + i - 1] = 0;
        }
    }
}
int main()
{
    cin >> n;
    dfs(1);
    cout << ans;
	return 0;
}
```
