### P1725 琪露诺
---

**难度**：$\mathit{\color{green}{普及+/\ 提高\ -}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 4e5 + 10;
const int INF = -(1 << 30);
int f[N], a[N];
int n, m; 
signed main()
{
    int l, r; cin >> n >> l >> r;
    for (int i = 0; i <= n; i ++)
        cin >> a[i];
    deque<int> dq;
    f[0] = 0;
    for (int i = 1; i <= n; i ++)
    {
        f[i] = INF;
        if (dq.size() && dq.front() < i - r)
            dq.pop_front();
        if (i - l >= 0)
        {
            while (dq.size() && f[dq.back()] <= f[i - l])
                dq.pop_back();
            dq.push_back(i - l);
        }
        if (dq.empty()) f[i] = INF;
        else f[i] = f[dq.front()] + a[i];
    }
    int ans = INF;
    for (int i = max(n - r + 1, 0); i <= n; i ++)
        ans = max(ans, f[i]);
    cout << ans;
	return 0;
}
```
