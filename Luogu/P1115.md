### P1115 最大子段和



```
#include<bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10;
int a[N], f[N];
signed main()
{
	cin.tie(0) -> sync_with_stdio(0);
	int n; cin >> n;
	for (int i = 1; i <= n; i ++)
		cin >> a[i];
    int ans = -1e9;
	for (int i = 1; i <= n; i ++)
	{
		f[i] = max(f[i - 1] + a[i], a[i]);
        ans = max(ans, f[i]);
	}
	cout << ans;
	return 0;
}
```
