### P2672 [NOIP 2015 普及组] 推销员
---

**难度**：$\mathit{\color{cyan}{提高}}$

解法一：优先队列
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
typedef pair<int, int> PII;
int a[N], s[N];
int main()
{
    int n; cin >> n;
    for (int i = 1; i <= n; i ++)
        cin >> s[i], s[i] *= 2;
    for (int i = 1; i <= n; i ++)
        cin >> a[i], a[i] *= 2;
    int mxn = -1, mxi, dis;
    for (int i = 1; i <= n; i ++)
    {
        if (a[i] + s[i] > mxn)
        {
            mxn = a[i] + s[i];
            mxi = i;
            dis = s[i];
        }
    }
    priority_queue<PII> heap;
    for (int i = 1; i <= n; i ++)
    {
        if (i < mxi)
            heap.push({a[i], -1});
        if (i > mxi)
            heap.push({a[i] + s[i] - dis, s[i]});
    }
    cout << mxn << '\n';
    int tired = a[mxi], walk = dis;
    while (heap.size())
    {
        auto t = heap.top(); heap.pop();
        int val = t.first, ss = t.second;
        if (ss != -1)//有距离贡献.
            val += dis - ss;//还原为a[i].
        if (walk >= ss && ss != -1)//已不存在距离贡献(归类错误)
        {
            heap.push({val, -1});//重新计算价值，继续下次循环重新计算
            continue;
        }
        //选择（对答案贡献最大的值）
        tired += val;
        walk = max(walk, ss);
        cout << tired + walk << '\n';
    }
    return 0;
}
```
解法二：线段树
```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N = 1e5 + 10;
int s[N], dis[N];
bool st[N];
struct Sgt
{
    int l, r;
    int dat, val, i;
} t[N*4];
void pushup(Sgt & r, Sgt & lson, Sgt & rson)
{
	if (lson.val >= rson.val)
		r.val = lson.val, r.i = lson.i;
	else
		r.val = rson.val, r.i = rson.i;
}
void pushup(int p)
{
	pushup(t[p], t[p*2], t[p*2+1]);
}
void build(int p, int l, int r)
{
	t[p].l = l, t[p].r = r;
	if (l == r)
	{
		t[p].i = l;
		return;
	}
	int mid = (l + r) >> 1;
	build(p*2, l, mid);
	build(p*2 + 1, mid + 1, r);
	pushup(p);
}
void change(int p, int x, int v)
{
	if (t[p].l == t[p].r)
	{
		t[p].i = t[p].l;
		t[p].dat = v;
		t[p].val = v;
		return;
	}
	int mid = (t[p].l + t[p].r) >> 1;
	if (x <= mid)
		change(p*2, x, v);
	else
		change(p*2 + 1, x, v);
	pushup(p);
}
void add(int p, int x, int v)
{
	if (t[p].l == t[p].r)
	{
		t[p].i = t[p].l;
		t[p].dat += v;
		t[p].val = t[p].dat;
		return;
	}
	int mid = (t[p].l + t[p].r) >> 1;
	if (x <= mid)
		add(p*2, x, v);
	else
		add(p*2 + 1, x, v);
	pushup(p);
}
Sgt query(int p, int l, int r)
{
	if (l <= t[p].l && t[p].r <= r)
		return t[p];
	int mid = (t[p].l + t[p].r) >> 1;
	if (r <= mid)
		return query(p*2, l, r);
	if (l > mid)
		return query(p*2 + 1, l, r);
	Sgt ans, lson = query(p*2, l, mid), rson = query(p*2 + 1, mid + 1, r);
	pushup(ans, lson, rson);
	return ans;
}
signed main()
{
    int n; cin >> n;
    for (int i = 1; i <= n; i++)
        cin >> dis[i];
    for (int i = 1; i <= n; i++)
        cin >> s[i];
    build(1, 1, n);
    for (int i = 1; i <= n; i++)
    	change(1, i, s[i] + dis[i] * 2);
    int walk = 0;
    int ans = 0;
	for (int i = 1; i <= n; i++)
	{
		auto ver = t[1];
		int pos = ver.i, ctb = ver.val;
		change(1, pos, -1e9);
		st[pos] = 1;
		if (dis[pos]*2 > walk)
		{
			ans -= walk;
			for (int j = 1; j <= pos; j++)
            {
                if (st[j] == 0)
                    change(1, j, s[j]);
            }
			for (int j = pos + 1; j <= n; j++)
                if (st[j] == 0)
                    add(1, j, -(dis[pos] * 2 - walk));
			walk = dis[pos]*2;
			ans += walk;
		}
		ans += s[pos];
        cout << ans << '\n';
	}
    return 0;
}
```
