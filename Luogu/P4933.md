### P4933 大师

**难度**：$\mathit{\color{green}{普及+/\ 提高\ -}}$

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N = 1e3 + 10, M = 3e4;
const int MOD = 998244353;
int f[N][M * 2 + 100];
int a[N];
int mod(int x)
{
	x %= MOD;
	if (x < 0) x += MOD;
	return x;
}
signed main()
{
	cin.tie(0) -> sync_with_stdio(0);
	int n; cin >> n;
	for (int i = 1; i <= n; i ++)
		cin >> a[i];
	for (int i = 1; i <= n; i ++)
	{
		for (int j = 1; j < i; j ++)
		{
			int sub = a[i] - a[j] + M, k = 0;
			f[i][sub] = mod(f[i][sub] + f[j][sub] + 1);
		}
	}
    int ans = n;
    for (int i = 1; i <= n; i ++)
        for (int j = 0; j <= 2 * M + 10; j ++)
            ans = mod(ans + f[i][j]);
	cout << ans;
	return 0;
}
```
