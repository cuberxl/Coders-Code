### P1020 [NOIP 1999 提高组] 导弹拦截
---

**难度**：$\mathit{\color{green}{普及+/\ 提高\ -}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
int a[N], q[N];
int main()
{
	cin.tie(0) -> sync_with_stdio(0);
	int n = 1;
    while (cin >> a[n]) n ++;
    n --;
	int len = 0;
	q[0] = -1e9;
	for (int i = 1; i <= n; i ++)
	{
		int l = 0, r = len;
		while (l < r)
		{
			int mid = (l + r + 1) >> 1;
			if (q[mid] >= a[i])
				l = mid;
			else
				r = mid - 1;
		}
		q[l + 1] = a[i];
		len = max(len, l + 1);
	}
	cout << len << '\n';
    
    len = 0;
	q[0] = -1e9;
	for (int i = 1; i <= n; i ++)
	{
		int l = 0, r = len;
		while (l < r)
		{
			int mid = (l + r + 1) >> 1;
			if (q[mid] < a[i])
				l = mid;
			else
				r = mid - 1;
		}
		q[l + 1] = a[i];
		len = max(len, l + 1);
	}
	cout << len;
	return 0;
}
```
