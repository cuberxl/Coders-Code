### P14362 [CSP-S 2025] 道路修复
---

**难度**：$\mathit{\color{cyan}{提高}}$

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N = 2e6 + 10;
const int INF = 0x3f3f3f3f3f3f3f3f;
int n, m, k, nm;
struct Edge
{
	int a, b, w;
	bool operator < (const Edge & o) const
	{
		return w < o.w;
	}
};
Edge eg[N];
int p[N], cs[N];
bool st[N];
int find(int x)
{
	if (p[x] != x) p[x] = find(p[x]);
	return p[x];
}
void solve()
{
	for (int i = 1; i <= n + k; i ++)
		p[i] = i;
	for (int i = 1; i <= m; i ++)
	{
		int a = eg[i].a, b = eg[i].b, w = eg[i].w;
		int ra = find(a), rb = find(b);
		if (ra != rb)
		{
			p[ra] = rb;
            eg[++ nm] = {a, b, w};
		}
	}
}
int kruskal(int chos)
{
	int ans = 0, cnt = 0;
    for (int i = 1; i <= k; i ++)
    {
        st[i] = chos & 1;
        chos >>= 1;
        if (st[i]) ans += cs[i];
    }
	for (int i = 1; i <= n + k; i ++)
		p[i] = i;
	for (int i = 1; i <= m; i ++)
	{
		int a = eg[i].a, b = eg[i].b, w = eg[i].w;
        if (a > n && !st[a - n]) continue;
		int ra = find(a), rb = find(b);
		if (ra != rb)
		{
			p[ra] = rb;
			ans += w;
            cnt ++;
		}
	}
    if (cnt < n - 1) return INF;
    else return ans;
}
signed main()
{
	cin.tie(0) -> sync_with_stdio(0);
	cin >> n >> m >> k;
	for (int i = 1; i <= m; i ++) 
	{
		int a, b, c; cin >> a >> b >> c;
		eg[i] = {a, b, c};
	}
	sort(eg + 1, eg + m + 1);
    solve();
    m = nm;
    for (int i = 1; i <= k; i ++) 
	{
        cin >> cs[i];
        for (int j = 1; j <= n; j ++)
        {
            int a; cin >> a;
            eg[++ m] = {i + n, j, a};
        }
	}
    sort(eg + 1, eg + m + 1);
    int ans = INF;
    for (int i = 0; i < 1 << k; i ++)
    {
        int t = kruskal(i);
        ans = min(ans, t);
    }
    cout << ans;
	return 0;	
}
```
