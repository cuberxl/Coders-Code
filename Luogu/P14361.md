### P14361 [CSP-S 2025] 社团招新
---

**难度**：$\mathit{\color{yellow}{普及}}$

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N = 1e5 + 10;
int a[N][10];
short b[N];
signed main()
{
    int t; cin >> t;
    while (t--)
    {
        int n; cin >> n;
        for (int i = 1; i <= n; i++)
            cin >> a[i][1] >> a[i][2] >> a[i][3];
        int st, nd, th;
        st = nd = th = 0;
        int ans = 0;
        for (int i = 1; i <= n; i ++)
        {
            if (a[i][1] >= max(a[i][2], a[i][3]))
                b[i] = 1, st ++ , ans += a[i][1];
            else if (a[i][2] >= max(a[i][1], a[i][3]))
                b[i] = 2, nd ++ , ans += a[i][2];
            else if (a[i][3] >= max(a[i][1], a[i][2]))
                b[i] = 3, th ++ , ans += a[i][3];
        }
        priority_queue<int, vector<int>, greater<int>> heap;
        for (int i = 1; i <= n; i ++)
        {
            if (st > n/2 && b[i] == 1)
                heap.push(a[i][1] - max(a[i][2], a[i][3]));
            else if (nd > n/2 && b[i] == 2)
                heap.push(a[i][2] - max(a[i][1], a[i][3]));
            else if (th > n/2 && b[i] == 3)
                heap.push(a[i][3] - max(a[i][1], a[i][2]));
        }
        while (st > n/2)
            ans -= heap.top(), heap.pop(), st --;
        while (nd > n/2)
            ans -= heap.top(), heap.pop(), nd --;
        while (th > n/2)
            ans -= heap.top(), heap.pop(), th --;
        cout << ans << '\n';
    }
    return 0;
}
```
