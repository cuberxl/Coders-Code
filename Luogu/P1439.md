### P1439 两个排列的最长公共子序列
---

**难度**：$\mathit{\color{green}{普及+/\ 提高\ -}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
int a[N], b[N], x[N], q[N];
int main()
{
	cin.tie(0) -> sync_with_stdio(0);
    int n; cin >> n;
    for (int i = 1; i <= n; i ++)
    {
        cin >> a[i];
        x[a[i]] = i;
    }
    for (int i = 1; i <= n; i ++)
    {
        cin >> b[i];
        b[i] = x[b[i]];
    }
    int len = 0;
    q[0] = -1e9;
    for (int i = 1; i <= n; i ++)
    {
        int l = 0, r = len;
        while (l < r)
        {
            int mid = (l + r + 1) >> 1;
            if (q[mid] < b[i])
                l = mid;
            else
                r = mid - 1;
        }
        q[l + 1] = b[i];
        len = max(len, l + 1);
    }
    cout << len;
	return 0;
}
```
