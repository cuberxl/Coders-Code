### P1948 [USACO08JAN] Telephone Lines S
---

**难度**：$\mathit{\color{green}{普及+/\ 提高\ -}}$


做法一：分层图最短路
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10, M = 2e4 + 10, INF = 0x3f3f3f3f;
typedef tuple<int, int, int> TP;
int n, m, k; 
int idx;
int h[N], to[M], ne[M], w[M];
int dis[N][N];
bool st[N][N];
void add(int a, int b, int c)
{
    ne[idx] = h[a];
    to[idx] = b;
    w[idx] = c;
    h[a] = idx ++;
}
void dijkstra()
{
    memset(dis, 0x3f, sizeof dis);
    dis[1][0] = 0;
    priority_queue<TP, vector<TP>, greater<TP>> heap;
    heap.push(make_tuple(0, 1, 0));
    while (heap.size())
    {
        auto t = heap.top(); heap.pop();
        int u = get<1>(t), p = get<2>(t);
        if (st[u][p]) continue;
        st[u][p] = 1;
        for (int i = h[u]; ~i; i = ne[i])
        {
            int j = to[i];
            if (dis[j][p] > max(dis[u][p], w[i]))//不免费该边：更新该路径的最大边权
            {
                dis[j][p] = max(dis[u][p], w[i]);
                heap.push(make_tuple(dis[j][p], j, p));
            }
            if (dis[j][p + 1] > dis[u][p] && p + 1 <= k)//免费该边：若该免费在限度内，用前一个点的最短路松弛
            {
                dis[j][p + 1] = dis[u][p];
                heap.push(make_tuple(dis[j][p + 1], j, p + 1));
            }
            
        }
    }
}
int main()
{
    memset(h, -1, sizeof h);
    cin >> n >> m >> k;
    while (m --)
    {
        int a, b, c;
        cin >> a >> b >> c;
        add(a, b, c);
        add(b, a, c);
    }
    dijkstra();
    int ans = INF;
    for (int i = 0; i <= k; i ++)
        ans = min(ans, dis[n][i]);
    if (ans == INF)
        ans = -1;
    cout << ans;
    return 0;
}
```

做法二：双端队列BFS

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10, M = 2e4 + 10, INF = 0x3f3f3f3f;
typedef tuple<int, int, int> TP;
int n, m, k; 
int idx;
int h[N], to[M], ne[M], w[M];
int dis[N];
bool st[N];
void add(int a, int b, int c)
{
    ne[idx] = h[a];
    to[idx] = b;
    w[idx] = c;
    h[a] = idx ++;
}
bool check(int x)
{
	memset(dis, 0x3f, sizeof dis);
	dis[1] = 0;
	memset(st, 0, sizeof st);
	deque<int> dq;
	dq.push_back(1);
	while (dq.size())
	{
		int u = dq.front(); dq.pop_front();
		if (st[u]) continue;
		st[u] = 1;
		for (int i = h[u]; ~i; i = ne[i])
		{
			int v = to[i];
			int d;
			if (w[i] > x) d = 1;
			else d = 0;
			if (dis[v] > dis[u] + d)
			{
				dis[v] = dis[u] + d;
				if (d == 1) dq.push_back(v);
				else dq.push_front(v);
			}
		}
	}
	if (dis[n] <= k)
		return 1;
	else
		return 0;
}
int main()
{
    memset(h, -1, sizeof h);
    cin >> n >> m >> k;
    while (m --)
    {
        int a, b, c;
        cin >> a >> b >> c;
        add(a, b, c);
        add(b, a, c);
    }
    int l = 0, r = 1e6 + 1;
	while (l < r)
	{
		int mid = (l + r) >> 1;
		if (check(mid)) r = mid;
		else l = mid + 1;
	}
	int ans = l;
	if (ans == 1e6 + 1)
	    ans = -1;
	cout << ans;
    return 0;
}
```
