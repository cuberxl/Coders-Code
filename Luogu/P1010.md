### P1010 [NOIP 1998 普及组] 幂次方
---

**难度**：$\mathit{\color{orange}{普及-}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e2 + 10;
string solve(int x)
{
	if (x == 1)
		return "2(0)";
	if (x == 2)
		return "2";
	if (x == 4)
		return "2(2)";
	string ans;
	int prod = 1;
	int p = 0;
	while (prod * 2 <= x)
		p++, prod *= 2;
	x -= prod;
	string ps;
	if (p == 0 || p == 2)
		ps += p + '0';
	else
		ps = solve(p);
	if (p != 1)
		ans += "2(" + ps + ")";
	else
		ans += "2";
	if (x)
	{
		ans += "+";
		ans += solve(x);
	}
	return ans;
}
signed main()
{
	ios::sync_with_stdio(0);
	cin.tie(0); cout.tie(0);
    int n;
    cin >> n;
    cout << solve(n);
	return 0 ;
}
```
