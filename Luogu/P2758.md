###  P2758 编辑距离
---

**难度**：$\mathit{\color{yellow}{普及}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 2e3 + 10;
int f[N][N];
int main()
{
	cin.tie(0) -> sync_with_stdio(0);
    string a, b;
    cin >> a >> b;
    memset(f, 0x3f, sizeof f);
    f[0][0] = 0;
    for (int i = 1; i <= b.size(); i ++)
        f[0][i] = f[i][0] = i;
	for (int i = 1; i <= a.size(); i ++)
        for (int j = 1; j <= b.size(); j ++)
        {
            f[i][j] = min(f[i][j], f[i - 1][j] + 1);//cha ru
            f[i][j] = min(f[i][j], f[i][j - 1] + 1);//shan chu
            if (a[i - 1] == b[j - 1])
                f[i][j] = min(f[i][j], f[i - 1][j - 1]);
            else
                f[i][j] = min(f[i][j],f[i - 1][j - 1] + 1);
        }
    cout << f[a.size()][b.size()];
	return 0;
}
```
