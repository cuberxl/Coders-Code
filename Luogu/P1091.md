

### P1115 最大子段和


**难度**：$\mathit{\color{orange}{普及-}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
int a[N], f[N], g[N];
int main()
{
	cin.tie(0) -> sync_with_stdio(0);
	int n; cin >> n;
	for (int i = 1; i <= n; i ++)
		cin >> a[i];
	for (int i = 1; i <= n; i ++)
	{
		f[i] = 1;
		for (int j = 1; j < i; j ++)
		{
			if (a[j] < a[i])
				f[i] = max(f[i], f[j] + 1);
		}
	}
	for (int i = n; i >= 1; i --)
	{
		g[i] = 1;
		for (int j = n; j > i; j --)
		{
			if (a[i] > a[j])
				g[i] = max(g[i], g[j] + 1);
		}
	}
	int ans = 0;
	for (int i = 1; i <= n; i ++)
		ans = max(ans, f[i] + g[i] - 1);
	cout << n - ans;
	return 0;
}#include<bits/stdc++.h>
```
