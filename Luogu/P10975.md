### P10975 Mondriaan's Dream / 蒙德里安的梦想
---

**难度**：$\mathit{\color{blue}{提高+/省选-}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=22,M=(1<<12)+10;
long long f[N][M];
bool st[M];
int main()
{
    int n,m;
	while (cin>>n>>m,n||m)
    {
        memset(f,0,sizeof f);
        for (int i=0;i<(1<<n);i++)
        {
            st[i]=1;
            int cnt=0;
            for (int j=0;j<n;j++)
            {
                if ((i>>j&1)==1)
                {
                    if (cnt&1)
                        st[i]=0;
                    cnt=0;
                }
                else
                    cnt++;
            }
            if (cnt&1)
                st[i]=0;
        }
        f[1][0]=1;
        for (int i=2;i<=m+1;i++)
            for (int j=0;j<(1<<n);j++)
                for (int k=0;k<(1<<n);k++)
                {
                    if ((j&k)==0&&st[j|k])
                        f[i][j]+=f[i-1][k];
                }
        cout<<f[m+1][0]<<'\n';
    }
	return 0;
}
```
