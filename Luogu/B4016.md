### B4016 树的直径
---

**难度**：$\mathit{\color{yellow}{普及}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10;
int n, m; 
int idx, ans = 1e9;
int h[N], to[N], ne[N];
int dis[N], f[N];
bool st[N];
void add(int a, int b)
{
    ne[idx] = h[a];
    to[idx] = b;
    h[a] = idx ++;
}
int dfs(int u, int fa)
{
    int mx = 0, smx = 0;
    for (int i = h[u]; ~i; i = ne[i])
    {
        int v = to[i];
        if (v == fa) continue;
        dfs(v, u);
        int d = dis[v] + 1;
        if (d > mx)
            smx = mx, mx = d;
        else if (d > smx)
            smx = d;
    }
    dis[u] = mx;
    f[u] = mx + smx;
    return f[u];
}
int main()
{
    memset(h, -1, sizeof h);
    cin >> n;
    for (int i = 1; i < n; i ++)
    {
        int u, v; cin >> u >> v;
        add(u, v);
        add(v, u);
    }
    dfs(1, 0);
    int ans = 0;
    for (int i = 1; i <= n; i ++)
        ans = max(ans, f[i]);
    cout << ans;
    return 0;
}
```
