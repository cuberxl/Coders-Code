### P1080 [NOIP 2012 提高组] 国王游戏
---

**难度**：$\mathit{\color{cyan}{提高}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
typedef long long ll;
typedef vector<int> VI;
struct S
{
	ll l, r, val;	
} s[N];
ll b[N];
bool cmp(S a, S b)
{
	return a.val < b.val;
}
void input(VI &a)
{
	string s; cin >> s;
	for (int i = s.size() - 1; i >= 0; i --)
		a.push_back(s[i] - '0');
}
void print(VI a)
{
	while (a.size())
	{
		cout << a.back();
		a.pop_back();
	}
}
bool compare(VI a, VI b)
{
	if (a.size() != b.size())
		return a.size() < b.size();
	for (int i = a.size() - 1; i >= 0; i --)
	{
		if (a[i] != b[i])
			return a[i] < b[i];
	}
	return 0;
}
VI mul(VI &a, int b)
{
	VI c; int x = 0;
	if (b == 0)
	{
		c.push_back(0);
		return c;
	}
	for (int i = 0; i < a.size() || x; i++)
	{
		if (i < a.size())
			x += a[i] * b;
		c.push_back(x % 10);
		x /= 10;
	}
	return c;
}
VI div(VI &a, int b)
{
	VI c; int x = 0;
	for (int i = a.size() - 1; i >= 0 ; i --)
	{
		x = x * 10 + a[i];
		c.push_back(x/b);
		x %= b;
	}
	reverse(c.begin(), c.end());
	while (c.size() && c.back() == 0)
		c.pop_back();
	return c;
}
signed main()
{
    ios::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
    ll n; cin >> n;
    VI x, y;
    input(x); input(y);
    for (int i = 1; i <= n; i ++)
    {
        cin >> s[i].l >> s[i].r;
        s[i].val = s[i].l * s[i].r;
    }
    sort(s + 1, s + n + 1, cmp);
    VI ans, prod = x;
    ans.push_back(0);
    for (int i = 1; i <= n; i ++)
    {
    	VI now = div(prod, s[i].r);
    	if (compare(ans,now))
    		ans = now;
    	prod = mul(prod, s[i].l);
	}
    print(ans);
    return 0;
}
```
