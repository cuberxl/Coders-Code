### P1158 [NOIP 2010 普及组] 导弹拦截
---

**难度**：$\mathit{\color{yellow}{普及}}$

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N = 1e5 + 10;
struct D
{
    int d_1, d_2;
    bool operator < (const D & other) const
    {
        return d_1 < other.d_1;
    }
} a[N];

signed main()
{
    int n, x_1, x_2, y_1, y_2;
    cin >> x_1 >> y_1 >> x_2 >> y_2;
    cin >> n;
    for (int i = 1; i <= n; i++)
    {
        int x, y;
        cin >> x >> y;
        a[i].d_1 = (x_1 - x) * (x_1 - x) + (y_1 - y) * (y_1 - y);
        a[i].d_2 = (x_2 - x) * (x_2 - x) + (y_2 - y) * (y_2 - y);
    }
    sort(a + 1, a + n + 1);
    int cir_2 = 0;
    int ans = a[n].d_1;
    for (int i = n - 1; i >= 0; i--)
    {
        cir_2 = max(cir_2, a[i + 1].d_2);
        ans = min(ans, a[i].d_1 + cir_2);
    }
    
    cout << ans;
    return 0;
}
```
