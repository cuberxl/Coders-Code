### P1434 [SHOI2002] 滑雪
---

**难度**：$\mathit{\color{yellow}{普及}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
int res = 1e9, ans;
int n, m; 
int s[10];
int f[N][N];
int a[N][N];
int dx[] = {1, 0, -1, 0}, dy[] = {0, 1, 0, -1};
int dfs(int x, int y)
{
    if (f[x][y] != -1)
        return f[x][y];
    for (int i = 0; i < 4; i ++)
    {
        int tx = x + dx[i], ty = y + dy[i];
        if (tx < 1 || ty < 1 || tx > n || ty > m)
            continue;
        if (a[tx][ty] < a[x][y])
            f[x][y] = max(f[x][y], dfs(tx, ty) + 1);
    }
    if (f[x][y] == -1)
    {
        f[x][y] = 0;
        return 0;
    }
    return f[x][y];
}
int main()
{
    cin >> n >> m;
    for (int i = 1; i <= n; i ++)
        for (int j = 1; j <= m; j ++)
            cin >> a[i][j];
    memset(f, -1, sizeof f);
    int ans = 0;
    for (int i = 1; i <= n; i ++)
        for (int j = 1; j <= m; j ++)
            ans = max(ans, dfs(i, j));
    cout << ans + 1;
	return 0;
}
```
