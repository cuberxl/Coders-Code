### P1352 没有上司的舞会
---

**难度**：$\mathit{\color{yellow}{普及}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 2e5 + 10;
int n, m; 
int idx, ans = 1e9;
int h[N], to[N], ne[N];
int r[N], f[N][5];
bool hf[N];
void add(int a, int b)
{
    ne[idx] = h[a];
    to[idx] = b;
    h[a] = idx ++;
}
void dfs(int u)
{
    f[u][1] = r[u];
    for (int i = h[u]; ~i; i = ne[i])
    {
        int v = to[i];
        dfs(v);
        f[u][0] += max(f[v][0], f[v][1]);
        f[u][1] += f[v][0];
    }
}
int main()
{
    memset(h, -1, sizeof h);
    cin >> n;
    for (int i = 1; i <= n; i ++)
        cin >> r[i];
    for (int i = 1; i < n; i ++)
    {
        int l, k; cin >> l >> k;
        add(k, l);
        hf[l] = 1;
    }
    int root = 1;
    while (hf[root]) root ++;
    dfs(root);
    cout << max(f[root][0], f[root][1]);
    return 0;
}
```
