### P2895 [USACO08FEB] Meteor Shower S
---

**难度**：$\mathit{\color{yellow}{普及}}$


```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10, INF = 0x3f3f3f3f;
typedef tuple<int, int, int> TP;
int reach[N][N];
int dx[] = {1, 0, -1, 0, 0}, dy[] = {0, 1, 0, -1, 0};
bool vis[N][N];
int bfs()
{
    queue<TP> q;
    q.push({0, 0, 0});
    vis[0][0] = 1;
    while (q.size())
    {
        auto t = q.front(); q.pop();
        int time = get<0>(t), x = get<1>(t), y = get<2>(t);
        if (reach[x][y] == INF)
            return time;
        for (int i = 0; i < 4; i ++)
        {
            int tx = x + dx[i], ty = y + dy[i];
            if (tx < 0 || ty < 0)
                continue;
            if (reach[tx][ty] > time + 1 && !vis[tx][ty])
            {
                vis[tx][ty] = 1;
                q.push({time + 1, tx, ty});
            }
        }
    }
    return -1;
}
int main()
{
    int m; cin >> m;
    memset(reach, 0x3f, sizeof reach);
    while (m --)
    {
        int x, y, t; cin >> x >> y >> t;
        for (int i = 0; i < 5; i ++)
        {
            int tx = x + dx[i], ty = y + dy[i];
            if (tx < 0 || ty < 0)
                continue;
            if (reach[tx][ty] > t)
                reach[tx][ty] = t;
        }
    }
    cout << bfs();
	return 0;
}
```
