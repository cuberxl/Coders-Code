### P1002 [NOIP 2002 普及组] 过河卒

**难度**：$\mathit{\color{orange}{普及-}}$

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N = 1e2 + 10;
int f[N][N];
bool st[N][N];
int dx[] = {-2, -2, -1, -1, 1, 1, 2, 2};
int dy[] = {1, -1, 2, -2, 2, -2, 1, -1};
int n, m, s, t;
void init()
{
	st[s][t] = 1;
	for (int i = 0; i < 8; i ++)
	{
		int tx = s + dx[i], ty = t + dy[i];
		if (0 <= tx && tx <= n && 0 <= ty && ty <= m)
			st[tx][ty] = 1;
	}
}
bool check(int x, int y)
{
	if (x < 0 || y < 0 || x > n || y > m)
		return 0;
	if (st[x][y])
		return 0;
	return 1;
}
signed main()
{
	ios::sync_with_stdio(0);
	cin.tie(0); cout.tie(0);
	cin >> n >> m;
	cin >> s >> t;
	f[0][0] = 1;
	init();
	for (int i = 0; i <= n; i ++)
	{
		for (int j = 0; j <= m; j ++)
		{
			if (i == 0 && j == 0 || st[i][j] == 1)
				continue;
			if (check(i - 1, j))
				f[i][j] += f[i - 1][j];
			if (check(i, j - 1))
				f[i][j] += f[i][j - 1];
		}
	}
	cout << f[n][m];
	return 0 ;
}
```
