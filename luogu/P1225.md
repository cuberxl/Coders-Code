### P1255 数楼梯
---

**难度**：$\mathit{\color{orange}{普及-}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 5e3 + 10;
typedef vector<short> VI;
VI f[N];
VI add(VI a, VI b)
{
	VI c;
	int x = 0;
	for (int i = 0; i < max(a.size(), b.size()); i ++)
	{
		if (i < a.size())
			x += a[i];
		if (i < b.size())
			x += b[i];
		c.push_back(x % 10);
		x /= 10; 
	}
	if (x)
		c.push_back(x); 
	while (c.size() && c.back() == 0)
		c.pop_back();
	return c;
}
void print(VI a)
{
	for (int i = a.size() - 1; i >= 0; i --)
		cout << a[i];
}
signed main()
{
	int n;
    cin >> n;
    f[0].push_back(1); 
    f[1].push_back(1); 
    for (int i = 2; i <= n; i++)
        f[i] = add(f[i-1], f[i-2]);
    print(f[n]);
	return 0 ;
}
```
