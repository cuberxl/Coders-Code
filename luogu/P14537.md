#include<iostream>
using namespace std;
int a[100];
int main()
{
    ios::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
    register char c=getchar();
    while (c!=EOF)
    {
        a[c]++;
        c=getchar();
    }
    for (register short i='9';i>='0';i--)
        for (register int j=1;j<=a[i];j++)
            putchar(i);
	return 0;
}
