### P1498 南蛮图腾
---

**难度**：$\mathit{\color{yellow}{普及}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
void print(int len, bool b)
{
    if (!b)
        return;
	while (len --)
		cout << ' ';
}
void solve(int size, int row, bool b)
{
	if (size == 1)
	{
		if (row == 1)
		{
			cout << "/\\";
		}
		if (row == 2)
		{
			cout << "/__\\";
		}
		return;
	}
	int m = pow(2, size);
	if (row <= m / 2)
	{
		print(m - row, b);
		solve(size - 1, row, 0);
		return;
	}
    else
    {
    	print(m - row, b);
    	solve (size - 1, row - m / 2, 0);
    	print(m * 2 - row * 2, 1);
    	solve (size - 1, row - m / 2, 0);
        return;
    }
}
int main()
{
	int n; cin >> n;
	int m = pow(2, n);
	for (int i = 1; i <= m; i ++)
	{
		solve(n, i, 1);
		cout << '\n';
	}
	return 0;
}
```
