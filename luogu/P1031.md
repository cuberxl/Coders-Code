#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N = 1e2 + 10;
int a[N];
signed main()
{
    ios::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
    int n;
    int avg = 0;
    cin >> n;
    for (int i = 1; i <= n; i++)
    {
        cin >> a[i];
        avg += a[i];
    }
    avg /= n;
    int ans = 0;
    for (int i = 1; i < n; i++)
    {
        if (a[i] != 0)
            ans++, a[i + 1] += a[i];
    }
    cout << ans;
    return 0;
}
