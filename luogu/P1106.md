#include <bits/stdc++.h>
using namespace std;
int main()
{
    ios::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
    string s; int k;
    cin >> s >> k;
    deque<char> stk;
    stk.push_back(s[0]);
    for (int i = 1; i < s.size(); i++)
    {
        while (k && stk.size() && stk.back() > s[i])
            stk.pop_back(), k--;
        stk.push_back(s[i]);
    }
    while (k>0)
        stk.pop_back(),k--;
    while (stk.size() && stk.front() == '0')
        stk.pop_front();
    if (stk.empty())
        cout<<'0';
    while (stk.size())
    {
        cout << stk.front();
        stk.pop_front();
    }
    return 0;
}
