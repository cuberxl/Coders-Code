### P2058 [NOIP 2016 普及组] 海港
---
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 3e5+10;
typedef pair<int,int> PII;
queue<PII> q;
int c[N];
int main()
{
    ios::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
	int n; cin >> n;
    int ans = 0;
	while (n--)
	{
		int t,k; cin>>t>>k;
		for (int i = 1; i <= k; i++)
        {
            int x; cin >> x;
            q.push({t,x});
            if (c[t] == 0)
                ans++;
            c[x]++;
        }
        while (q.size() && t - q.front().first >= 86400)
        {
            auto t = q.front(); q.pop();
            c[t.second]--;
            if (c[t.second] == 0)
                ans--;
        }
        cout << ans << '\n';
	}
    return 0;
}
```
