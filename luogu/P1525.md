### P1525 [NOIP 2010 提高组] 关押罪犯
---
**难度**：$\mathit{\color{green}{普及+/\ 提高\ -}}$
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=4e4+10,M=1e5+10;
int p[N];
struct Edge
{
    int a,b,c;
} ed[M];
bool cmp(Edge a,Edge b)
{
    return a.c>b.c;
}
int find(int x)
{
    if (p[x]!=x)
        p[x]=find(p[x]);
    return p[x];
}
void merge(int a,int b)
{
    int ra=find(a),rb=find(b);
    if (ra!=rb)
        p[ra]=rb;
}
int main()
{
    int n,m;
    cin>>n>>m;
    for (int i=1;i<=n*2;i++)
        p[i]=i;
    for (int i=1;i<=m;i++)
    {
        int a,b,c;
        cin>>a>>b>>c;
        ed[i]={a,b,c};
    }
    sort(ed+1,ed+m+1,cmp);
    for (int i=1;i<=m;i++)
    {
        int a=ed[i].a,b=ed[i].b,c=ed[i].c;
        // cout<<a<<' '<<b<<' '<<c<<'\n';
        if (find(a)==find(b))
        {
            cout<<c;
            return 0;
        }
        merge(a+n,b);
        merge(a,b+n);
    }
    cout<<0;
	return 0;
}
```
