### P1443 马的遍历
---

**难度**：$\mathit{\color{orange}{普及-}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
typedef pair<int, int> PII;
int n, m, sx, sy;
int f[N][N];
bool vis[N][N];
int dx[]={-2, -2, -1, -1, 1, 1, 2, 2}, dy[]={1, -1, 2, -2, 2, -2, 1, -1};
void bfs()
{
    queue<PII> q;
    q.push({sx, sy});
    memset(f, -1, sizeof f);
    f[sx][sy] = 0;
    while (q.size())
    {
        auto t = q.front(); q.pop();
        int x = t.first, y = t.second;
        for (int i = 0; i < 8; i ++)
        {
            int tx = x + dx[i], ty = y + dy[i];
            if (tx < 1 || tx > n || ty < 1 || ty > m)
                continue;
            if (f[tx][ty] == -1)
            {
                f[tx][ty] = f[x][y] + 1;
                q.push({tx, ty});
            }
        }
    }
}
int main()
{
    cin >> n >> m >> sx >> sy;
    bfs();
    for (int i = 1; i <= n; i ++)
    {
        for (int j = 1; j <= m; j ++)
            cout << f[i][j] << ' ';
        cout << '\n';
    }
    return 0;
 }
```
