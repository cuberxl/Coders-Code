做法二：动态规划
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
int n, k, ans;
int f[N][N];//i分j
int main()
{
    cin >> n >> k;
    for (int i = 1; i <= n; i ++)
        f[i][1] = 1;
    for (int i = 2; i <= n; i ++)
        for (int j = 1; j <= min(k, n); j ++)
        {
            if (i < j)
                f[i][j] = 0;
            f[i][j] = f[i - 1][j - 1] + f[i - j][j];
        }
    cout << f[n][k];
	return 0;
}
```
