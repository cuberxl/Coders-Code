### P1025 [NOIP 2001 提高组] 数的划分
---

**难度**：$\mathit{\color{yellow}{普及}}$

做法一：DFS
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
int n, k, ans;
int a[N];
void dfs(int u, int lst, int sum)
{
    if (u > k)
    {
        if (sum == n)
            ans ++;
        return;
    }
    if (sum >= n && u <= k)
        return;
    for (int i = lst; i <= (n - sum) / (k - u + 1); i ++)
        dfs(u + 1, i, sum + i);
}
int main()
{
    cin >> n >> k;
    dfs(1, 1, 0);
    cout << ans;
	return 0;
}
```

做法二：动态规划

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
int n, k, ans;
int f[N][N];//i分j
int main()
{
    cin >> n >> k;
    for (int i = 1; i <= n; i ++)
        f[i][1] = 1;
    for (int i = 2; i <= n; i ++)
        for (int j = 1; j <= min(k, n); j ++)
        {
            if (i < j)
                f[i][j] = 0;
            f[i][j] = f[i - 1][j - 1] + f[i - j][j];
        }
    cout << f[n][k];
	return 0;
}
```
