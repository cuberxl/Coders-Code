### P5767 [NOI1997] 最优乘车
---

**难度**：$\mathit{\color{yellow}{普及}}$

一、BFS
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 5e5 + 10, INF = 0x3f3f3f3f;
typedef pair<int, int> PII;
int n, m, idx;
int h[N], to[N], ne[N];
int w[N], dis[N];
void add(int a, int b, int c)
{
    ne[idx] = h[a];
    to[idx] = b;
    w[idx] = c;
    h[a] = idx ++;
}
int bfs()
{
    memset(dis, 0x3f, sizeof dis);
    dis[1] = 0;
    queue<int> q;
    q.push(1);
    while (q.size())
    {
    	int t = q.front(); q.pop();
    	for (int i = h[t]; i != -1; i = ne[i])
    	{
    		int j = to[i];
    		if (dis[j] != INF)
    			continue;
    		dis[j] = dis[t] + 1;
    		q.push(j);
		}
	}
	if (dis[n] == INF)
		return -1;
	else
	    return dis[n];
}
int main()
{
	memset(h, -1, sizeof h);
    cin >> m >> n;
    cin.ignore();
    while (m --)
    {
        string s;
        getline(cin, s);
        stringstream ss;
        ss << s;
        vector<int> a;
        int num; 
        while (ss >> num)
        {
        	for (int i = a.size() - 1; i >= 0; i--)
        		add(a[i], num, 1);
            a.push_back(num);
        }
    }
    int t = bfs();
    if (t == -1)
    	cout << "NO";
    else
    	cout << t - 1;
    return 0;
}
```
