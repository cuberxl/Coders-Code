### P1135 奇怪的电梯
---

**难度**：$\mathit{\color{yellow}{普及}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
int n, a, b;
int f[N], d[N];
void bfs()
{
    queue<int> q;
    q.push(a);
    memset(f, -1, sizeof f);
    f[a] = 0;
    while (q.size())
    {
        int t = q.front(); q.pop();
        if (f[t + d[t]] == -1 && (1 <= t + d[t] && t + d[t] <= n))
        {
            f[t + d[t]] = f[t] + 1;
            q.push(t + d[t]);
        }
        if (f[t - d[t]] == -1 && (1 <= t - d[t] && t - d[t] <= n))
        {
            f[t - d[t]] = f[t] + 1;
            q.push(t - d[t]);
        }
    }
}
int main()
{
    cin >> n >> a >> b;
    for (int i = 1; i <= n; i ++)
        cin >> d[i];
    bfs();
    cout << f[b];
    return 0;
 }
```
