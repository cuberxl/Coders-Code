```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
int res = 1e9, ans;
int s[10];
int v[N], f[N];
int main()
{
    cin >> s[1] >> s[2] >> s[3] >> s[4];
    int ans = 0;
    for (int i = 1; i <= 4; i ++)
    {
        memset(f, 0, sizeof f);
    	int m = 0;
        for (int j = 1; j <= s[i]; j ++)
    	{
    		cin >> v[j];
    		m += v[j];
		}
        for (int j = 1; j <= s[i]; j ++)
        	for (int k = m / 2; k >= v[j]; k --)
        		f[k] = max(f[k], f[k - v[j]] + v[j]);
        ans += max(f[m / 2], m - f[m / 2]);
	} 
    cout << ans;
	return 0;
}
```
