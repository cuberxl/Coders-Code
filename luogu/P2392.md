### P2392 kkksc03 考前临时抱佛脚
---

**难度**：$\mathit{\color{orange}{普及-}}$

方法一：DFS
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
int res = 1e9, ans;
int s[10];
int f[10][N]; 
void dfs(int p, int u, int lb, int rb)
{
    // printf("%d %d %d %d %d\n", res, p, u, lb, rb);
    if (u > s[p])
    {
        int x = max(lb, rb);
        res = min(res, x);
        return;
    }
    dfs(p, u + 1, lb + f[p][u], rb);
    dfs(p, u + 1, lb, rb + f[p][u]);
}
int main()
{
    cin >> s[1] >> s[2] >> s[3] >> s[4];
    for (int i = 1; i <= 4; i ++)
        for (int j = 1; j <= s[i]; j ++)
            cin >> f[i][j];
    for (int i = 1; i <= 4; i ++)
    {
        res = 1e9;
        dfs(i, 1, 0, 0);
        ans += res;
    }
    cout << ans;
	return 0;
}
```

方法二：01背包DP

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
int res = 1e9, ans;
int s[10];
int v[N], f[N];
int main()
{
    cin >> s[1] >> s[2] >> s[3] >> s[4];
    int ans = 0;
    for (int i = 1; i <= 4; i ++)
    {
        memset(f, 0, sizeof f);
    	int m = 0;
        for (int j = 1; j <= s[i]; j ++)
    	{
    		cin >> v[j];
    		m += v[j];
		}
        for (int j = 1; j <= s[i]; j ++)
        	for (int k = m / 2; k >= v[j]; k --)
        		f[k] = max(f[k], f[k - v[j]] + v[j]);
        ans += max(f[m / 2], m - f[m / 2]);
	} 
    cout << ans;
	return 0;
}
```
