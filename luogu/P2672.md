#include<bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
int s[N], dis[N], a[N];
bool st[N];
struct Sgt
{
    int l, r;
    int dat, val, i;
} t[N*4];
void pushup(Sgt & r, Sgt & lson, Sgt & rson)
{
	if (lson.val >= rson.val)
		r.val = lson.val, r.i = lson.i;
	else
		r.val = rson.val, r.i = rson.i;
}
void pushup(int p)
{
	pushup(t[p], t[p*2], t[p*2+1]);
}
void build(int p, int l, int r)
{
	t[p].l = l, t[p].r = r;
	if (l == r)
	{
		t[p].i = l;
		t[p].val = a[l];
		t[p].dat = a[l];
		return;
	}
	int mid = (l + r) >> 1;
	build(p*2, l, mid);
	build(p*2 + 1, mid + 1, r);
	pushup(p);
}
void change(int p, int x, int v)
{
	if (t[p].l == t[p].r)
	{
		t[p].i = t[p].l;
		t[p].dat = v;
		t[p].val = v;
		return;
	}
	int mid = (t[p].l + t[p].r) >> 1;
	if (x <= mid)
		change(p*2, x, v);
	else
		change(p*2 + 1, x, v);
	pushup(p);
}
void add(int p, int x, int v)
{
	if (t[p].l == t[p].r)
	{
		t[p].i = t[p].l;
		t[p].dat += v;
		t[p].val = t[p].dat;
		return;
	}
	int mid = (t[p].l + t[p].r) >> 1;
	if (x <= mid)
		change(p*2, x, v);
	else
		change(p*2 + 1, x, v);
	pushup(p);
}
Sgt query(int p, int l, int r)
{
	if (l <= t[p].l && t[p].r <= r)
		return t[p];
	int mid = (t[p].r + t[p].r) >> 1;
	if (r <= mid)
		return query(p*2, l, r);
	if (l > mid)
		return query(p*2 + 1, l, r);
	Sgt ans, lson = query(p*2, l, r), rson = query(p*2 + 1, l, r);
	pushup(ans, lson, rson);
	return ans;
}
int main()
{
    int n; cin >> n;
    for (int i = 1; i <= n; i++)
        cin >> dis[i];
    for (int i = 1; i <= n; i++)
        cin >> s[i];
    build(1, 1, n);
    for (int i = 1; i <= n; i++)
    	change(1, i, s[i] + dis[i] * 2);
    int walk = 0;
    int ans = 0;
	for (int i = 1; i <= n; i++)
	{
		auto ver = query(1, 1, n);
		int pos = ver.i, ctb = ver.val;
		change(1, pos, -1e9);
		st[pos] = 1;
		if (dis[pos]*2 > walk)
		{
			ans -= walk;
			for (int j = walk / 2 + 1; j <= pos; j++)
			{
				if (st[j] == 0)
					change(1, j, s[j]*2);
			}
			for (int j = pos + 1; j <= n; j++)
				add(1, j, - (dis[pos]*2 - walk));
			walk = dis[pos]*2;
			ans += walk;
		}
		ans += s[pos];
        cout << ans << '\n';
	}
    return 0;
}
}
