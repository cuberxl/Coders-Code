### P2672 [NOIP 2015 普及组] 推销员
---
**难度**：$\mathit{\color{cyan}{提高}}$

解法一：优先队列
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e5 + 10;
typedef pair<int, int> PII;
int a[N], s[N];
int main()
{
    int n; cin >> n;
    for (int i = 1; i <= n; i ++)
        cin >> s[i], s[i] *= 2;
    for (int i = 1; i <= n; i ++)
        cin >> a[i], a[i] *= 2;
    int mxn = -1, mxi, dis;
    for (int i = 1; i <= n; i ++)
    {
        if (a[i] + s[i] > mxn)
        {
            mxn = a[i] + s[i];
            mxi = i;
            dis = s[i];
        }
    }
    priority_queue<PII> heap;
    for (int i = 1; i <= n; i ++)
    {
        if (i < mxi)
            heap.push({a[i], -1});
        if (i > mxi)
            heap.push({a[i] + s[i] - dis, s[i]});
    }
    cout << mxn << '\n';
    int tired = a[mxi], walk = dis;
    while (heap.size())
    {
        auto t = heap.top(); heap.pop();
        int val = t.first, ss = t.second;
        if (ss != -1)//有距离贡献.
            val += dis - ss;//还原为a[i].
        if (walk >= ss && ss != -1)//已不存在距离贡献(归类错误)
        {
            heap.push({val, -1});//重新计算价值，继续下次循环重新计算
            continue;
        }
        //选择（对答案贡献最大的值）
        tired += val;
        walk = max(walk, ss);
        cout << tired + walk << '\n';
    }
    return 0;
}
```
