### P1605 迷宫
---
**难度**：$\mathit{\color{orange}{普及-}}$
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
int n, m, t, ans;
int sx, sy, fx, fy;
bool vis[N][N];
int dx[]={1,0,-1,0}, dy[]={0,1,0,-1};
void dfs(int x, int y)
{
    vis[x][y] = 1;
    if (x == fx && y == fy)
    {
        ans ++;
        return;
    }
    for (int i = 0 ; i < 4 ; i ++)
    {
        int tx = x + dx[i], ty = y + dy[i];
        if (tx < 1 || tx > n || ty < 1 || ty > m)
            continue;
        if (!vis[tx][ty])
        {
            vis[tx][ty] = 1;
            dfs(tx, ty);
            vis[tx][ty] = 0;
        }
    }
    vis[x][y] = 0;
}
signed main()
{
    cin >> n >> m >> t;
    cin >> sx >> sy >> fx >> fy;
    while (t--)
    {
        int x, y; cin >> x >> y;
        vis[x][y] = 1;
    }
    dfs(sx, sy);
    cout << ans;
    return 0;
 }
```
