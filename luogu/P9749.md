### P9749 [CSP-J 2023] 公路

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N=1e5+10;
int v[N],a[N],s[N];
signed main()
{
    ios::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
    int n,d;
    cin>>n>>d;
    for (int i=1;i<n;i++)
        cin>>v[i],s[i]=s[i-1]+v[i];
    s[n]=s[n-1];
    for (int i=1;i<=n;i++)
        cin>>a[i];
    int cost=0,pos=1;
    double oil=0;
    while (pos<=n)
    {
        // cout<<pos<<' '<<cost<<' '<<oil<<'\n';
        int k=pos+1;
        while (k<=n&&a[k]>a[pos])//find next cheap station
            k++;
        int ss=s[k-1]-s[pos-1];//the s to next
        double buy_oil=ss*1.0/d-oil;
        if (buy_oil>1e-6)//buy oil
            oil+=ceil(buy_oil),cost+=ceil(buy_oil)*a[pos];
        // cout<<s[k-1]<<' '<<cost<<' '<<oil<<'\n';
        pos=k;//reach next
        oil-=ss*1.0/d;//spend oil;
    }
    cout<<cost;
    return 0;
}
```
