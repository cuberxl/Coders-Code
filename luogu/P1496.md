###  P1496 火烧赤壁
---
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 2e4+10;
struct Range
{
    int l, r;
    bool operator < (const Range & others) const
    {
        return l < others.l;
    }
} rg[N];
int main()
{
	int n;
    cin >> n;
    for (int i = 1; i <= n; i++)
    {
        int l, r;
        cin >> l >> r;
        rg[i] = {l, r};
    }
    sort(rg + 1, rg + n + 1);
    int ans = rg[1].r - rg[1].l, led=rg[1].r;
    for (int i = 2; i <= n; i++)
    {
        if (rg[i].l >= led)
            ans += rg[i].r - rg[i].l;
        else if (rg[i].r > led)
            ans += rg[i].r - led;
        led = max(led, rg[i].r);
    }
    cout << ans;
	return 0;
}
```
