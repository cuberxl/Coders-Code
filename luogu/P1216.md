### P1216 [IOI1994] 数字三角形 Number Triangles
---

**难度**：$\mathit{\color{orange}{普及-}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
int a[N][N], f[N][N];
int main()
{
    int n; cin >> n;
    for (int i = 1; i <= n; i ++)
        for (int j = 1; j <= i; j ++)
            cin >> a[i][j];
    memset(f, 0xaf, sizeof f);
    f[1][1] = a[1][1];
    for (int i = 2; i <= n; i ++)
        for (int j = 1; j <= i; j ++)
            f[i][j] = max(f[i - 1][j - 1], f[i - 1][j]) + a[i][j];
    int ans = -1;
    for (int i = 1; i <= n; i ++)
        ans = max(ans, f[n][i]);
    cout << ans;
    return 0;
 }
```
