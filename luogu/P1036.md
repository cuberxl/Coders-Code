### P1036 [NOIP 2002 普及组] 选数
---

**难度**：$\mathit{\color{orange}{普及-}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e2 + 10;
int n, k, ans = 0; 
int a[N];
bool st[N];
bool prime(int x)
{
	if (x <= 1)
		return 0;
	for (int i = 2; i <= x/i; i++)
	{
		if (x % i == 0)
			return 0;
	}
	return 1;
}
void dfs(int u, int sum, int last)
{
	if (u > k)
	{
		if (prime(sum))
			ans ++;
		return;
	}
	for (int i = last + 1; i <= n; i ++)
	{
		if (!st[i])
		{
			st[i] = 1;
			dfs(u + 1, sum + a[i], i);
			st[i] = 0;
		}
	}
}
signed main()
{
	ios::sync_with_stdio(0);
	cin.tie(0); cout.tie(0);
    cin >> n >> k;
    for (int i = 1; i <= n; i ++)
    	cin >> a[i];
    dfs(1, 0, 0);
    cout << ans;
	return 0 ;
}
```
