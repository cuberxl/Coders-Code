### P1036 [NOIP 2002 普及组] 选数
---

**难度**：$\mathit{\color{orange}{普及-}}$

解法一：不降原则 + 上下界剪枝

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
int n, k, ans;
int a[N];
bool prime(int x)
{
    if (x <= 1)
        return 0;
    for (int i = 2; i <= x/i; i++)
    {
        if (x % i == 0)
            return 0;
    }
    return 1;
}
void dfs(int u, int lst, int sum)
{
    if (u > k)
    {
        if (prime(sum))
            ans ++;
        return;
    }
    for (int i = lst + 1; i <= n - k + u; i ++)
        dfs(u + 1, i, sum + a[i]);
}
int main()
{
    cin >> n >> k;
    for (int i = 1; i <= n; i ++)
        cin >> a[i];
    dfs(1, 0, 0);
    cout << ans;
	return 0;
}
}
```
