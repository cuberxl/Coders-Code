
### P7075 [CSP-S 2020]
---

**难度**：$\mathit{\color{orange}{普及-}}$

策略：打表，构造。

容易写出一个简单的DP程序，暴力算出小规模的解。
```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N=1e5+10,INF=LLONG_MAX;
int many[]={6,2,5,5,4,5,6,3,7,6};
int f[N];//i根小木棍摆f[i];
signed main()
{
    ios::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
    int t; cin>>t;
    for (int i=1;i<=N;i++)
        f[i]=INF;
    for (int i=1;i<=9;i++)
        f[many[i]]=min(f[many[i]],i);
    for (int i=1;i<=N;i++)
    {
        for (int j=0;j<=9;j++)
        {
            if (i-many[j]<0)
                continue;
            if (f[i-many[j]]==INF)
                continue;
            // cout<<f[i]<<' '<<f[i-many[j]]*10+j<<'\n';
            if (f[i-many[j]]!=0)
                f[i]=min(f[i],f[i-many[j]]*10+j);
        }
    }
    while (t--)
    {
        int n; cin>>n;
        if (f[n]!=INF)
            cout<<f[n]<<'\n';
        else
            cout<<-1<<'\n';
    }
    return 0;
}
```

然而，从中提取一些性质，可以发现每个数对于 $\mod 7$ 呈现出某种规律。根据这个性质，直接根据 $\mod 7$ 进行分类讨论。得出以下AC代码：
```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N=1e5+10,INF=LLONG_MAX;
int spj[]={-1,-1,1,7,4,2,6,8,10,18,22,20,28,68,88};
int f[N];//i根小木棍摆f[i];
signed main()
{
    ios::sync_with_stdio(0);
    cin.tie(0); cout.tie(0);
    int t; cin>>t;
    while (t--)
    {
        int n; cin>>n;
        if (n<=14)
        {
            cout<<spj[n]<<'\n';
            continue;
        }
        int cnt=n/7,mod=n%7;
        if (mod==1)
        {
            cout<<"10";
            cnt-=1;
        }
        else if (mod==2)
            cout<<"1";
        else if (mod==3)
        {
            cout<<"200";
            cnt-=2;
        }
        else if (mod==4)
        {
            cout<<"20";
            cnt-=1;
        }
        else if (mod==5)
            cout<<"2";
        else if (mod==6)
            cout<<"6";
        if (cnt<0)
            continue;
        else
        {
            while (cnt--)
                cout<<8;
        }
        cout<<'\n';
    }
    return 0;
}
```
