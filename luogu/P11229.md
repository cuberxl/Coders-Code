
### P7075 [CSP-S 2020] 儒略日

**难度**：$\mathit{\color{orange}{普及-}}$

策略：打表，构造。

容易写出一个简单的DP程序，暴力算出小规模的解。

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int A=146097,B=1461;
int y[A],m[A],d[A];
int m_d(int y,int m)
{
    if (m==2)
    {
        if ((y%4==0&&y%100!=0)||(y%400==0))
            return 29;
        else
            return 28;
    }
    if (m==4||m==6||m==9||m==11)
        return 30;
    return 31;
}
signed main()
{
    m[0]=d[0]=1;
    for (int i=1;i<A;i++)
    {
        d[i]=d[i-1]+1,m[i]=m[i-1],y[i]=y[i-1];
        if (d[i]>m_d(y[i],m[i]))
            d[i]=1,m[i]++;
        if (m[i]>12)
            m[i]=1,y[i]++;
    }
    int t;
    cin>>t;
    while (t--)
    {
        int r; cin>>r;
        int base_y=0;
        if (r>2299160)
        {
            r-=2159351;
            base_y=1200+r/A*400;
            r%=A;
        }
        else
        {
            base_y=-4712+r/B*4;
            r%=B;
        }
        int ans_y=base_y+y[r],ans_m=m[r],ans_d=d[r];
        if (ans_y>=0)
            cout<<ans_d<<' '<<ans_m<<' '<<ans_y<<'\n';
        else
            cout<<ans_d<<' '<<ans_m<<' '<<1-ans_y<<" BC"<<'\n';
    }
    return 0;
}
```
