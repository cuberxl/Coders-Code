### P1255 数楼梯
---

**难度**：$\mathit{\color{orange}{普及-}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 5e3 + 10;
typedef vector<int> VI;
struct Num
{
	VI d;
    Num() {};
	Num (int x)
	{
		if (x == 0)
		{
			d.push_back(0);
			return;
		}
		while (x)
		{
			d.push_back(x % 10);
			x /= 10;
		}
	}
	Num operator + (const Num & o) const
	{
		Num c;
		int x = 0, len = max(d.size(), o.d.size());
		for (int i = 0; i < len || x; i ++)
		{
			if (i < d.size())
				x += d[i];
			if (i < o.d.size())
				x += o.d[i];
			c.d.push_back(x % 10);
			x /= 10; 
		}
		while (c.d.size() > 1 && c.d.back() == 0)
			c.d.pop_back();
		return c;
	}
	friend ostream & operator << (ostream & os, const Num & b)
	{
		for (int i = b.d.size() - 1; i >= 0; i--)
			os << b.d[i];
		return os;
	}
};
Num f[N];
signed main()
{
	int n;
    cin >> n;
    f[1] = Num(1);
    f[2] = Num(2);
    for (int i = 3; i <= n; i++)
        f[i] = f[i - 1] + f[i - 2];
    cout << f[n];
	return 0 ;
}
```
