### P1928 外星密码
---

**难度**：$\mathit{\color{yellow}{普及}}$

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
string solve()
{
	string ans;
	char c;
	while (cin >> c)
	{
		if (c == '[')
		{
			int k; cin >> k;
			string zip = solve();
			while (k --)
				ans += zip;
		}
		else if (c == ']')
			return ans;
		else
			ans += c;
	}
    return ans;
}
signed main()
{
	ios::sync_with_stdio(0);
	cin.tie(0); cout.tie(0);
    cout << solve();
	return 0 ;
}
```
