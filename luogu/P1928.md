### P1928 外星密码
---
```cpp
#include<bits/stdc++.h>
using namespace std;
const int N = 1e3 + 10;
string s;
string solve()
{
	string ans;
	char c = getchar();
	while (c != EOF)
	{
		if (c == '[')
		{
			int k; cin >> k;
			string zip = solve();
			while (k --)
				ans += zip;
		}
		else if (c == ']')
			return ans;
		else
			s += c;
	}
}
signed main()
{
	ios::sync_with_stdio(0);
	cin.tie(0); cout.tie(0);
    cout << solve();
	return 0 ;
}
```
