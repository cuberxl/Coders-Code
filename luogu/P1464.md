### P1464 [PacNW 1999] Function
---

**难度**：$\mathit{\color{orange}{普及-}}$

```cpp
#include<bits/stdc++.h>
#define int long long
using namespace std;
const int N = 50 + 10, INF = LLONG_MAX;
int f[N][N][N]; 
int w(int a, int b, int c)
{
	if (a <= 0 || b <= 0 || c <= 0)
		return 1;
	if (a > 20 || b > 20 || c > 20)
		return w(20, 20, 20);
	if (f[a][b][c] != INF)
		return f[a][b][c];
	if (a < b && b < c)
		f[a][b][c] = w(a, b, c - 1) + w(a, b - 1, c - 1) - w(a, b - 1, c);
	f[a][b][c] = w(a - 1, b, c) + w (a - 1, b - 1, c) + w(a - 1, b, c - 1) - w(a - 1, b - 1, c - 1);
	return f[a][b][c];
}
signed main()
{
    int a, b, c;
    for (int i = 1; i <= 20; i ++)
    for (int j = 1; j <= 20; j ++)
    for (int k = 1; k <= 20; k ++)
    	f[i][j][k] = INF;
    w(20, 20, 20);
    while (cin >> a >> b >> c)
    {
    	if (a == -1 && b == -1 && c == -1)
			return 0;
		int ans;
		if (a <= 0 || b <= 0 || c <= 0)
			ans = 1;
		else if (a > 20 || b > 20 || c > 20)
			ans = f[20][20][20];
		else
			ans = w(a, b, c);
		printf("w(%lld, %lld, %lld) = %lld\n", a, b ,c, ans);
	}
	return 0 ;
}
```
